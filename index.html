// ClimaPocketApp.kt - Jetpack Compose 1.7 / Material 3.5
@Composable
fun ClimaPocketScreen(
    viewModel: WeatherViewModel = hiltViewModel()
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val dynamicColors = rememberDynamicAgroScheme(
        temp = uiState.currentStation.temperature,
        humidity = uiState.currentStation.humidity
    )

    MaterialTheme(colorScheme = dynamicColors) {
        Scaffold(
            topBar = { AgroTopAppBar(uiState.currentStation) },
            bottomBar = { M3AgroNavigationBar() }
        ) { paddingValues ->
            LazyColumn(
                modifier = Modifier
                    .fillMaxSize()
                    .padding(paddingValues),
                contentPadding = PaddingValues(16.dp),
                verticalArrangement = Arrangement.spacedBy(14.dp)
            ) {
                item {
                    HeroWeatherCard(
                        station = uiState.currentStation,
                        et0 = calculateFao56Et0(uiState.currentStation)
                    )
                }
                item {
                    DiurnalForecastRow(uiState.diurnalForecast)
                }
                item {
                    AgronomicAdvisoryCard(
                        advisory = uiState.currentAdvisory,
                        driftRisk = uiState.sprayWindowRisk
                    )
                }
            }
        }
    }
}
